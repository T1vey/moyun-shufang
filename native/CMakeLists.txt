cmake_minimum_required(VERSION 3.13)
project(tcb705 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
add_library(tcb_core SHARED tcb_planner/core.cpp)
set_target_properties(tcb_core PROPERTIES PREFIX "lib")
if(WIN32)
  set_target_properties(tcb_core PROPERTIES PREFIX "")
endif()
add_executable(tcb_runner huacheng_bridge/runner.cpp)
find_package(Threads REQUIRED)
target_link_libraries(tcb_runner PRIVATE Threads::Threads)
option(TCB_WITH_ARM32_SDK "Build vendor SDK read-only probe on ARM32 Linux" OFF)
if(TCB_WITH_ARM32_SDK)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4 OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^arm")
    message(FATAL_ERROR "SDK binary requires Linux ARM32 armhf")
  endif()
  add_executable(tcb_sdk_probe huacheng_bridge/sdk_probe.cpp huacheng_bridge/sdk_adapter.cpp)
  set_target_properties(tcb_sdk_probe PROPERTIES CXX_STANDARD 17 BUILD_RPATH "$ORIGIN/../../vendor/huacheng/armhf")
  target_include_directories(tcb_sdk_probe PRIVATE ../vendor/huacheng/include)
  target_link_libraries(tcb_sdk_probe PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../vendor/huacheng/armhf/libxoip.so Threads::Threads)
endif()
